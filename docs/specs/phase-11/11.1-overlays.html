Part A: Overlay system (src/components/overlay/)

dismissal.ts

export type DismissReason = 'backdrop' | 'escape' | 'button';
export type DismissResult = 'close' | 'confirm' | 'ignore';

export interface DismissState {
  dirty: boolean;
  busy: boolean;
  dismissible: boolean;
}

/** Pure decision table for every overlay close attempt. */
export function resolveDismissal(reason: DismissReason, state: DismissState): DismissResult {
  if (state.busy) return 'ignore';
  if (reason !== 'button' && !state.dismissible) return 'ignore';
  return state.dirty ? 'confirm' : 'close';
}

overlayStack.ts

/** Tracks open overlays so only the topmost reacts to Escape, and locks page scroll while any is open. */
const stack: string[] = [];
let previousOverflow = '';

export function pushOverlay(id: string): void {
  if (stack.length === 0 && typeof document !== 'undefined') {
    previousOverflow = document.documentElement.style.overflow;
    // eslint-disable-next-line no-restricted-syntax -- runtime: page scroll lock
    document.documentElement.style.overflow = 'hidden';
  }
  stack.push(id);
}

export function removeOverlay(id: string): void {
  const index = stack.lastIndexOf(id);
  if (index >= 0) stack.splice(index, 1);
  if (stack.length === 0 && typeof document !== 'undefined') {
    // eslint-disable-next-line no-restricted-syntax -- runtime: page scroll lock
    document.documentElement.style.overflow = previousOverflow;
  }
}

export function isTopOverlay(id: string): boolean {
  return stack[stack.length - 1] === id;
}

Overlay.tsx

import React, { useEffect, useId, useRef, useState } from 'react';
import { createPortal } from 'react-dom';
import { cn } from '../ui';
import { resolveDismissal, type DismissReason } from './dismissal';
import { pushOverlay, removeOverlay, isTopOverlay } from './overlayStack';
import { DiscardChangesDialog } from './DiscardChangesDialog';

export interface OverlayRenderApi {
  /** Use for every X and Cancel button. Applies the dirty/busy rules. */
  requestClose: () => void;
}

export interface OverlayProps {
  open: boolean;
  /** Actually closes the overlay (parent sets its open state to false). */
  onClose: () => void;
  variant?: 'modal' | 'drawer';
  dirty?: boolean;
  busy?: boolean;
  dismissible?: boolean;
  'aria-label'?: string;
  'aria-labelledby'?: string;
  panelClassName?: string;
  /** Runtime-only styles (e.g. tier-themed gradients). Never static values. */
  panelStyle?: React.CSSProperties;
  panelTestId?: string;
  children: React.ReactNode | ((api: OverlayRenderApi) => React.ReactNode);
}

const FOCUSABLE = 'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])';

export const Overlay: React.FC<OverlayProps> = props => (props.open ? <OverlayInner {...props} /> : null);

const OverlayInner: React.FC<OverlayProps> = ({
  onClose,
  variant = 'modal',
  dirty = false,
  busy = false,
  dismissible = true,
  panelClassName,
  panelStyle,
  panelTestId,
  children,
  ...aria
}) => {
  const id = useId();
  const panelRef = useRef<HTMLDivElement>(null);
  const pressStartedOnBackdrop = useRef(false);
  const [confirmOpen, setConfirmOpen] = useState(false);

  const requestClose = (reason: DismissReason) => {
    const result = resolveDismissal(reason, { dirty, busy, dismissible });
    if (result === 'close') onClose();
    if (result === 'confirm') setConfirmOpen(true);
  };
  const requestCloseRef = useRef(requestClose);
  requestCloseRef.current = requestClose;

  // Stack registration, focus capture/restore.
  useEffect(() => {
    const previouslyFocused = document.activeElement as HTMLElement | null;
    pushOverlay(id);
    const panel = panelRef.current;
    const target = panel?.querySelector<HTMLElement>('[data-autofocus]') ?? panel;
    target?.focus();
    return () => {
      removeOverlay(id);
      previouslyFocused?.focus?.();
    };
  }, [id]);

  // Escape: only the topmost overlay reacts.
  useEffect(() => {
    const onKeyDown = (e: KeyboardEvent) => {
      if (e.key !== 'Escape' || !isTopOverlay(id)) return;
      e.preventDefault();
      requestCloseRef.current('escape');
    };
    document.addEventListener('keydown', onKeyDown);
    return () => document.removeEventListener('keydown', onKeyDown);
  }, [id]);

  // Warn before tab close or reload while there are unsaved changes.
  useEffect(() => {
    if (!dirty) return;
    const onBeforeUnload = (e: BeforeUnloadEvent) => {
      e.preventDefault();
      e.returnValue = '';
    };
    window.addEventListener('beforeunload', onBeforeUnload);
    return () => window.removeEventListener('beforeunload', onBeforeUnload);
  }, [dirty]);

  const trapTab = (e: React.KeyboardEvent) => {
    if (e.key !== 'Tab' || !panelRef.current) return;
    const items = Array.from(panelRef.current.querySelectorAll<HTMLElement>(FOCUSABLE));
    if (items.length === 0) return;
    const first = items[0];
    const last = items[items.length - 1];
    if (e.shiftKey && document.activeElement === first) {
      e.preventDefault();
      last.focus();
    } else if (!e.shiftKey && document.activeElement === last) {
      e.preventDefault();
      first.focus();
    }
  };

  const content = (
    <div
      className={cn(
        'fixed inset-0 z-1000 flex bg-black/75 backdrop-blur-overlay animate-fade-in',
        variant === 'drawer' ? 'justify-end' : 'items-center justify-center p-4'
      )}
      onMouseDown={e => {
        pressStartedOnBackdrop.current = e.target === e.currentTarget;
      }}
      onClick={e => {
        if (pressStartedOnBackdrop.current && e.target === e.currentTarget) requestClose('backdrop');
        pressStartedOnBackdrop.current = false;
      }}
    >
      <div
        ref={panelRef}
        role="dialog"
        aria-modal="true"
        {...aria}
        tabIndex={-1}
        data-testid={panelTestId}
        onKeyDown={trapTab}
        // eslint-disable-next-line no-restricted-syntax -- runtime: caller-supplied themed panel styles
        style={panelStyle}
        className={cn(
          'relative outline-none',
          variant === 'drawer'
            ? 'flex h-full w-full max-w-[520px] flex-col overflow-hidden animate-slide-in-right'
            : 'max-h-[calc(100vh-2rem)] w-full overflow-y-auto animate-fade-in',
          panelClassName
        )}
      >
        {typeof children === 'function' ? children({ requestClose: () => requestClose('button') }) : children}
      </div>
      <DiscardChangesDialog
        open={confirmOpen}
        onKeepEditing={() => setConfirmOpen(false)}
        onDiscard={() => {
          setConfirmOpen(false);
          onClose();
        }}
      />
    </div>
  );

  return typeof document === 'undefined' ? content : createPortal(content, document.body);
};

The typeof document === 'undefined' branch keeps renderToStaticMarkup tests working, since they run in Node without a DOM.

ConfirmDialog.tsx

import React, { useId } from 'react';
import { AlertTriangle, Info, X } from 'lucide-react';
import { Button, IconButton, cn } from '../ui';
import { Overlay } from './Overlay';

export interface ConfirmDialogProps {
  open: boolean;
  title: string;
  body: React.ReactNode;
  tone?: 'danger' | 'warning' | 'info';
  confirmLabel: string;
  confirmVariant?: 'danger' | 'primary';
  cancelLabel?: string;
  onConfirm: () => void;
  onCancel: () => void;
  busy?: boolean;
  confirmDisabled?: boolean;
  /** Extra buttons rendered between Cancel and Confirm. */
  extraActions?: React.ReactNode;
}

const TONE = {
  danger: { border: 'border-red/50', icon: 'text-red' },
  warning: { border: 'border-gold', icon: 'text-gold-bright' },
  info: { border: 'border-line', icon: 'text-cyan' },
} as const;

export const ConfirmDialog: React.FC<ConfirmDialogProps> = ({
  open,
  title,
  body,
  tone = 'danger',
  confirmLabel,
  confirmVariant = 'danger',
  cancelLabel = 'Cancel',
  onConfirm,
  onCancel,
  busy = false,
  confirmDisabled = false,
  extraActions,
}) => {
  const titleId = useId();
  const Icon = tone === 'info' ? Info : AlertTriangle;
  return (
    <Overlay
      open={open}
      onClose={onCancel}
      busy={busy}
      aria-labelledby={titleId}
      panelClassName={cn('max-w-[460px] overflow-hidden rounded-lg border bg-surface shadow-lg', TONE[tone].border)}
    >
      {({ requestClose }) => (
        <>
          <div className="flex items-center justify-between gap-3 border-b border-line bg-elevated px-6 py-5">
            <div className="flex items-center gap-2.5">
              <Icon size={20} className={TONE[tone].icon} />
              <h3 id={titleId} className="text-lg font-bold text-fg">{title}</h3>
            </div>
            <IconButton aria-label="Close" onClick={requestClose} disabled={busy}>
              <X size={18} />
            </IconButton>
          </div>
          <div className="flex flex-col gap-4 p-6 text-sm text-fg">{body}</div>
          <div className="flex flex-wrap justify-end gap-3 border-t border-line bg-elevated px-6 py-4">
            <Button variant="secondary" onClick={requestClose} disabled={busy} data-autofocus>
              {cancelLabel}
            </Button>
            {extraActions}
            <Button variant={confirmVariant} onClick={onConfirm} disabled={busy || confirmDisabled}>
              {confirmLabel}
            </Button>
          </div>
        </>
      )}
    </Overlay>
  );
};

/** Equal if identical after dropping trailing empty games (Best-of changes pad with empties). */
function gamesEqual(a: GameDraft[], b: GameDraft[]): boolean {
  const trim = (list: GameDraft[]) => {
    const out = [...list];

Part C: How to convert one overlay

  1. Keep the component's exported props unchanged. The only exception is OrgMetadataModal's new isDirty.
  2. Keep the existing null guard (for example if (!isOpen || !match) return null;) exactly as it is, and render <Overlay open …>. The guard already handles closed state.
  3. Delete the outer fixed backdrop <div>. The panel <div> (its first child) becomes the Overlay panel: convert its static styles into panelClassName using the conversion guide. Width, background, border, radius, shadow and overflow go there. Runtime-themed values go in panelStyle.
  4. Delete onClick={e => e.stopPropagation()} from the panel.
  5. Use the render-prop form {({ requestClose }) => (…)}. Change every X button and every Cancel/Close/Back-out button from onClose (or onCancel) to requestClose. Save, Confirm and other action buttons keep their handlers.
  6. Give the title element an id from useId() and pass aria-labelledby.
  7. Delete any Escape-key useEffect in the component.
  8. Leave the interior content's inline styles alone. The 11.2 batches convert them. Exception: ConfirmDialog bodies are new JSX, so write them with Tailwind classes.
