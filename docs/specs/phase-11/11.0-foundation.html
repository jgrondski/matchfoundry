3c. scripts/captureVisualFixtures.ts

/**
 * Captures frozen API responses for the visual regression suite.
 * Requires the dev server (npm run dev) and SYSTEM_ADMIN_RECOVERY_PIN in .env.
 * Usage: npm run visual:capture -- [slug]   (default slug: omen-ao3)
 */
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

if (!process.env.SYSTEM_ADMIN_RECOVERY_PIN) {
  try {
    process.loadEnvFile();
  } catch {
    // No .env file; rely on the shell environment
  }
}

const BASE_URL = process.env.VISUAL_CAPTURE_BASE_URL || 'http://localhost:5173';
const SLUG = process.argv[2] || 'omen-ao3';
const PIN = (process.env.SYSTEM_ADMIN_RECOVERY_PIN || '').trim();
const OUT_DIR = path.join(path.dirname(fileURLToPath(import.meta.url)), '..', 'e2e', 'visual', 'fixtures');
const SECRET_KEY = /pin|token|secret|webhook/i;
const EXTRA_PLAYERS = 40;

function scrub(value: unknown): unknown {
  if (Array.isArray(value)) return value.map(scrub);
  if (value && typeof value === 'object') {
    const out: Record<string, unknown> = {};
    for (const [key, val] of Object.entries(value)) {
      if (SECRET_KEY.test(key)) continue;
      out[key] = scrub(val);
    }
    return out;
  }
  return value;
}

async function getJson(urlPath: string, headers: Record<string, string> = {}): Promise<any> {
  const res = await fetch(`${BASE_URL}${urlPath}`, { headers });
  if (!res.ok) throw new Error(`GET ${urlPath} failed with ${res.status}`);
  return res.json();
}

function write(name: string, data: unknown): void {
  fs.writeFileSync(path.join(OUT_DIR, `${name}.json`), `${JSON.stringify(scrub(data), null, 2)}\n`);
  console.log(`  wrote fixtures/${name}.json`);
}

async function main(): Promise<void> {
  if (!PIN) throw new Error('SYSTEM_ADMIN_RECOVERY_PIN is not set.');
  fs.mkdirSync(OUT_DIR, { recursive: true });

  const tournament = await getJson(`/api/tournaments/${SLUG}`);
  if (!tournament?.isLocked || !tournament.tiers?.some((t: any) => t.bracket)) {
    throw new Error(`Tournament "${SLUG}" must be locked and have a generated bracket.`);
  }

  const loginRes = await fetch(`${BASE_URL}/api/auth/pin`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ pin: PIN }),
  });
  const login = await loginRes.json();
  if (!loginRes.ok || !login.token) throw new Error('System admin login failed.');
  const auth = { Authorization: `Bearer ${login.token}` };

  const tournamentJson = JSON.stringify(tournament);
  const players: any[] = await getJson('/api/players');
  const keptPlayers = players.filter((p, i) => i < EXTRA_PLAYERS || tournamentJson.includes(p.id));

  write('health', await getJson('/api/health'));
  write('tournaments', await getJson('/api/tournaments'));
  write('tournament', tournament);
  write('organizations', await getJson('/api/organizations'));
  write('players', keptPlayers);
  write('sessions', await getJson(`/api/tournaments/${tournament.id}/qualify/sessions`, auth));
}

main().catch(err => {
  console.error(err);
  process.exit(1);
});

3d. e2e/visual/mockApi.ts

import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import type { Page } from '@playwright/test';

const FIXTURE_DIR = path.join(path.dirname(fileURLToPath(import.meta.url)), 'fixtures');

function load(name: string): any {
  return JSON.parse(fs.readFileSync(path.join(FIXTURE_DIR, `${name}.json`), 'utf8'));
}

const fixtures = {
  health: load('health'),
  tournaments: load('tournaments'),
  tournament: load('tournament'),
  organizations: load('organizations'),
  players: load('players'),
  sessions: load('sessions'),
};

export const FIXTURE_SLUG: string = fixtures.tournament.slug;
export const FIXTURE_TIER_SLUG: string = [...fixtures.tournament.tiers].sort(
  (a: any, b: any) => a.priority - b.priority
)[0].slug;

export const ADMIN_SESSION = {
  token: 'visual-test-token',
  role: 'SYSTEM_ADMIN',
  tournamentIds: [],
  tournamentSlugs: [],
  expiresAt: 4102444800000,
};

/** Routes every /api/* request to frozen fixtures. Returns a list that collects unmocked GETs. */
export async function installApiMocks(page: Page): Promise<string[]> {
  const unmocked: string[] = [];
  await page.route(
    url => url.pathname.startsWith('/api/'),
    async route => {
      const request = route.request();
      const { pathname } = new URL(request.url());
      if (request.method() !== 'GET') {
        return route.fulfill({ status: 200, json: { success: true } });
      }
      if (pathname === '/api/health') return route.fulfill({ json: fixtures.health });
      if (pathname === '/api/tournaments') return route.fulfill({ json: fixtures.tournaments });
      if (pathname === '/api/players') return route.fulfill({ json: fixtures.players });
      if (pathname === '/api/organizations') return route.fulfill({ json: fixtures.organizations });
      if (/^\/api\/tournaments\/[^/]+\/qualify\/sessions$/.test(pathname)) {
        return route.fulfill({ json: fixtures.sessions });
      }
      const single = pathname.match(/^\/api\/tournaments\/([^/]+)$/);
      if (single && (single[1] === fixtures.tournament.slug || single[1] === fixtures.tournament.id)) {
        return route.fulfill({ json: fixtures.tournament });
      }
      unmocked.push(`GET ${pathname}`);
      return route.fulfill({ status: 404, json: { error: `visual mock: unmocked GET ${pathname}` } });
    }
  );
  return unmocked;
}

Button.tsx

import React from 'react';
import { cn } from './cn';

export type ButtonVariant = 'primary' | 'secondary' | 'danger' | 'ghost';
export type ButtonSize = 'sm' | 'md';

const BASE =
  'inline-flex items-center justify-center font-sans font-semibold rounded-sm border border-transparent ' +
  'cursor-pointer no-underline whitespace-nowrap transition-all duration-150 ease-in-out ' +
  'disabled:cursor-not-allowed disabled:opacity-60';

const VARIANTS: Record<ButtonVariant, string> = {
  primary:
    'bg-linear-135 from-gold-bright to-gold text-canvas font-bold shadow-gold-sm ' +
    'enabled:hover:brightness-110 enabled:hover:shadow-gold-hover enabled:hover:-translate-y-px',
  secondary: 'bg-elevated border-line text-fg enabled:hover:bg-highlight enabled:hover:border-fg-secondary',
  danger: 'bg-red-soft border-red text-red-300 enabled:hover:bg-red enabled:hover:text-white',
  ghost: 'bg-transparent text-fg-secondary enabled:hover:bg-white/5 enabled:hover:text-fg',
};

const SIZES: Record<ButtonSize, string> = {
  sm: 'gap-1.5 px-3 py-1.5 text-xs',
  md: 'gap-2 px-4 py-2 text-sm',
};

/** Class string for elements that must look like a button but aren't <button> (e.g. router <Link>). */
export function buttonClasses(opts: { variant?: ButtonVariant; size?: ButtonSize; className?: string } = {}): string {
  const { variant = 'secondary', size = 'md', className } = opts;
  return cn(BASE, VARIANTS[variant], SIZES[size], className);
}

export interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: ButtonVariant;
  size?: ButtonSize;
}

export const Button = React.forwardRef<HTMLButtonElement, ButtonProps>(
  ({ variant = 'secondary', size = 'md', className, type = 'button', ...props }, ref) => (
    <button ref={ref} type={type} className={buttonClasses({ variant, size, className })} {...props} />
  )
);
Button.displayName = 'Button';

The enabled:hover: modifier works on <button>. For a <Link> using buttonClasses, those hover styles don't apply, because a link is never :enabled. Pass the plain hover equivalents in className instead (for secondary: hover:bg-highlight hover:border-fg-secondary). The conversion guide lists these.

import React from 'react';

export interface ChipProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {

  /** Runtime accent (e.g. a tier color). textColor must contrast with color. */
  accent?: { color: string; textColor: string };
  size?: 'sm' | 'md';
}

/** Pill-shaped toggle: tier selectors, view toggles, filter pills, nav pills. */
export const Chip = React.forwardRef<HTMLButtonElement, ChipProps>(
  ({ active = false, accent, size = 'md', className, style, type = 'button', ...props }, ref) => {
    const accentStyle = accent
      ? ({ '--chip-accent': accent.color, '--chip-accent-fg': accent.textColor } as React.CSSProperties)
      : undefined;
    return (
      <button
        ref={ref}
        type={type}
        aria-pressed={active}
        style={accentStyle ? { ...accentStyle, ...style } : style}
        className={cn(
          'inline-flex items-center gap-2 rounded-full border font-semibold transition-all duration-150',
          size === 'md' ? 'px-3.5 py-1.5 text-caption' : 'px-2.5 py-1 text-xs',

          active && !accent && 'bg-gold-soft text-gold-bright border-gold shadow-gold-glow',
          active &&
            accent &&
            'bg-(--chip-accent) text-(--chip-accent-fg) border-(--chip-accent) shadow-[0_0_10px_color-mix(in_srgb,var(--chip-accent)_25%,transparent)]',
          className
        )}
        {...props}
      />
    );
  }
);

useMediaQuery.ts

import { useCallback, useSyncExternalStore } from 'react';

/** Matches Tailwind's md breakpoint: anything below 768px is mobile. */
export const MOBILE_QUERY = '(max-width: 767.98px)';

export function useMediaQuery(query: string): boolean {
  const subscribe = useCallback(
    (onChange: () => void) => {
      if (typeof window === 'undefined' || !window.matchMedia) return () => {};
      const mql = window.matchMedia(query);
      mql.addEventListener('change', onChange);
      return () => mql.removeEventListener('change', onChange);
    },
    [query]
  );
  return useSyncExternalStore(
    subscribe,
    () => typeof window !== 'undefined' && !!window.matchMedia && window.matchMedia(query).matches,
    () => false
  );
}

export function useIsMobile(): boolean {
  return useMediaQuery(MOBILE_QUERY);
}
