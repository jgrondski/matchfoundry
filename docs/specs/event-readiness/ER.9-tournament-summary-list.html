Specs / Event readiness / ER.9

Step ER.9: Load a summary list, not every tournament in full

GET /api/tournaments returns one small summary per tournament in a fixed 5 queries. The full tournament is loaded only for the page being viewed. Pages show their loading screen until that tournament arrives.

Executor: any model. Prerequisite: ER.4 merged. Run before ER.6 (both edit src/features/tournament/store.tsx and src/server/api.ts). Can run in parallel with ER.10 (no shared files). Read first: index.html “Rules for every executor”, docs/TESTING_STRATEGY.md.

Why

Today GET /api/tournaments calls listFullTournaments(): every tournament ever created, each with all tiers, matches, games, qualifier scores, runs and players (about 8 queries per tournament). The store calls it on every page load for every visitor, including spectators opening one bracket link, and the response is not cached. On the preview, 4 tournaments cost 33 queries and 338 KB per page load, and the cost grows with every past and test tournament. On event day hundreds of spectators open the link at once.

After this step the list is a fixed 5 queries regardless of how many tournaments exist, and each page loads only its own tournament (App.tsx already calls refreshTournament(slug) for the current route).

Behavior after this step

  • The front page, tournament switcher, admin login, top nav count, player directory and organization pages look and behave as before.
  • Opening any tournament page shows that page's existing loading screen until the tournament has loaded, then renders as before. A slug that does not exist still shows the page's existing “not found” state.
  • The OBS hub page (/:slug/obs) now shows a loading screen while loading instead of briefly showing “Tournament Not Found”.
  • The organization dashboard loads its own tournaments in full from a new endpoint for its metrics.

A2. src/features/tournament/summary.ts (new)

import type { MatchScoreRecord, Tournament, TournamentSummary } from './types';

/** True when any match has a result or a recorded game. Same rule the tournament card used before ER.9. */
export function hasMatchProgress(matchScores: Record<string, MatchScoreRecord> | null | undefined): boolean {
  return Object.values(matchScores || {}).some(
    m =>
      m.isComplete ||
      m.player1Wins > 0 ||
      m.player2Wins > 0 ||
      Boolean(m.games?.some(g => g.player1Points !== null || g.player2Points !== null))
  );
}

/** Summary of a fully loaded tournament, identical in shape to what GET /api/tournaments returns. */
export function toTournamentSummary(t: Tournament): TournamentSummary {
  const playerIds = new Set<string>();
  (t.playersPool || []).forEach(p => playerIds.add(p.id));
  Object.keys(t.tournamentPlayers || {}).forEach(id => playerIds.add(id));
  (t.qualifierSubmissions || []).forEach(s => playerIds.add(s.playerId));
  (t.qualRuns || []).forEach(r => playerIds.add(r.playerId));
  return {
    id: t.id,
    slug: t.slug,
    name: t.name,
    organizationId: t.organizationId,
    date: t.date || '',
    location: t.location || '',
    qualFormat: t.qualFormat,
    qualAverageCount: t.qualAverageCount,
    pointsConfig: t.pointsConfig,
    qualMode: t.qualMode || 'HYBRID',
    onlineQualsOpen: Boolean(t.onlineQualsOpen),
    qualsClosed: Boolean(t.qualsClosed),
    isLocked: Boolean(t.isLocked),
    seedingMethod: t.seedingMethod || 'QUALIFIERS',
    isTest: Boolean(t.isTest),
    revision: typeof t.revision === 'number' ? t.revision : 0,
    tiers: (t.tiers || []).map(tier => ({
      id: tier.id,
      slug: tier.slug,
      name: tier.name,
      priority: tier.priority,
      playerCount: tier.playerCount,
      bestOf: tier.bestOf,
      primaryColor: tier.primaryColor,
    })),
    registeredPlayerCount: (t.playersPool || []).length,
    hasRecordedResults:
      (t.qualifierSubmissions || []).length > 0 ||
      (t.qualifiers || []).length > 0 ||
      hasMatchProgress(t.matchScores),
    playerIds: [...playerIds],
  };
}

/**
 * The list the UI shows: the server's summaries in server order, with each tournament that is
 * loaded in full replaced by its fresh summary, and loaded tournaments missing from the server
 * list (just created or simulated) first, in their loaded order.
 */
export function mergeTournamentSummaries(
  serverSummaries: TournamentSummary[],
  loaded: Tournament[]
): TournamentSummary[] {
  const loadedById = new Map(loaded.map(t => [t.id, t]));
  const serverIds = new Set(serverSummaries.map(s => s.id));
  const added = loaded.filter(t => !serverIds.has(t.id)).map(toTournamentSummary);
  const merged = serverSummaries.map(s => {
    const full = loadedById.get(s.id);
    return full ? toTournamentSummary(full) : s;
  });
  return [...added, ...merged];
}

If a field used above does not exist on Tournament or TournamentTier under that exact name (for example qualRuns, qualifiers, revision), stop and ask. Do not rename fields.

/** One summary per tournament, newest first, in 5 queries regardless of tournament count. */
export async function listTournamentSummaries(): Promise<TournamentSummary[]> {
  const db = getDb();
  const rows = await db
    .select({
      id: tournaments.id,
      slug: tournaments.slug,
      name: tournaments.name,
      organizationId: tournaments.organizationId,
      qualFormat: tournaments.qualFormat,
      qualAverageCount: tournaments.qualAverageCount,
      pointsConfig: tournaments.pointsConfig,
      qualMode: tournaments.qualMode,
      onlineQualsOpen: tournaments.onlineQualsOpen,
      metaOrganizationId: sql<string | null>`${tournaments.metadata} -> 'organizationId'`,
      metaDate: sql<string | null>`${tournaments.metadata} -> 'date'`,
      metaLocation: sql<string | null>`${tournaments.metadata} -> 'location'`,
      metaQualMode: sql<string | null>`${tournaments.metadata} -> 'qualMode'`,
      metaSeedingMethod: sql<string | null>`${tournaments.metadata} -> 'seedingMethod'`,
      metaIsLocked: sql<boolean | null>`${tournaments.metadata} -> 'isLocked'`,
      metaIsTest: sql<boolean | null>`${tournaments.metadata} -> 'isTest'`,
      metaRevision: sql<number | null>`${tournaments.metadata} -> 'revision'`,
      metaMatchScores: sql<Record<string, MatchScoreRecord> | null>`${tournaments.metadata} -> 'matchScores'`,
    })
    .from(tournaments)
    .orderBy(desc(tournaments.createdAt));
  if (rows.length === 0) return [];

  const [tierRows, enrolledRows, submissionRows, runRows] = await Promise.all([
    db
      .select({
        id: bracketTiers.id,
        tournamentId: bracketTiers.tournamentId,
        priorityOrder: bracketTiers.priorityOrder,
        name: bracketTiers.name,
        slug: bracketTiers.slug,
        numPlayers: bracketTiers.numPlayers,
        primaryColor: bracketTiers.primaryColor,
        bestOf: sql<number | null>`${bracketTiers.metadata} -> 'bestOf'`,
      })
      .from(bracketTiers)
      .orderBy(bracketTiers.priorityOrder),
    db.select({ tournamentId: tournamentPlayers.tournamentId, playerId: tournamentPlayers.playerId }).from(tournamentPlayers),
    db.select({ tournamentId: qualifierSubmissions.tournamentId, playerId: qualifierSubmissions.playerId }).from(qualifierSubmissions),
    db.select({ tournamentId: qualifierRuns.tournamentId, playerId: qualifierRuns.playerId }).from(qualifierRuns),
  ]);

  const group = <T extends { tournamentId: string }>(list: T[]) => {
    const byId = new Map<string, T[]>();
    for (const row of list) {
      const bucket = byId.get(row.tournamentId);
      if (bucket) bucket.push(row);
      else byId.set(row.tournamentId, [row]);
    }
    return byId;
  };
  const tiersBy = group(tierRows);
  const enrolledBy = group(enrolledRows);
  const submissionsBy = group(submissionRows);
  const runsBy = group(runRows);

  return rows.map(t => {
    const enrolled = enrolledBy.get(t.id) || [];
    const submissions = submissionsBy.get(t.id) || [];
    const runs = runsBy.get(t.id) || [];
    const playerIds = new Set<string>();
    for (const r of [...enrolled, ...submissions, ...runs]) playerIds.add(r.playerId);
    const isLocked = Boolean(t.metaIsLocked);
    return {
      id: t.id,
      slug: t.slug,
      name: t.name,
      organizationId: t.organizationId || t.metaOrganizationId || undefined,
      date: t.metaDate || '',
      location: t.metaLocation || '',
      qualFormat: t.qualFormat as TournamentSummary['qualFormat'],
      qualAverageCount: t.qualAverageCount || undefined,
      pointsConfig: (t.pointsConfig as TournamentSummary['pointsConfig']) || undefined,
      qualMode: ((t.qualMode || t.metaQualMode || 'HYBRID') as TournamentSummary['qualMode']),
      onlineQualsOpen: Boolean(t.onlineQualsOpen),
      // Same rule as getFullTournament: quals close exactly when brackets lock.
      qualsClosed: isLocked,
      isLocked,
      seedingMethod: ((t.metaSeedingMethod || 'QUALIFIERS') as TournamentSummary['seedingMethod']),
      isTest: Boolean(t.metaIsTest),
      revision: typeof t.metaRevision === 'number' ? t.metaRevision : 0,
      tiers: (tiersBy.get(t.id) || []).map(tier => ({
        id: tier.id,
        slug: tier.slug || tier.id,
        name: tier.name,
        priority: tier.priorityOrder,
        playerCount: tier.numPlayers,
        bestOf: typeof tier.bestOf === 'number' ? tier.bestOf : 3,
        primaryColor: tier.primaryColor,
      })),
      registeredPlayerCount: enrolled.length,
      hasRecordedResults: submissions.length > 0 || hasMatchProgress(t.metaMatchScores),
      playerIds: [...playerIds],
    };
  });
}

/** Every tournament of one organization, in full, newest first (organization dashboard metrics). */
export async function listFullTournamentsForOrganization(organizationId: string): Promise<Tournament[]> {
  const all = await listTournamentSummaries();
  const ids = all.filter(s => s.organizationId === organizationId).map(s => s.id);
  const full = await Promise.all(ids.map(id => getFullTournament(id)));
  return full.filter((t): t is Tournament => Boolean(t));
}
